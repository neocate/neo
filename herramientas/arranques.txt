# Ver procesos python de herramientas/ corriendo (PID + linea de comando)
Get-CimInstance Win32_Process -Filter "Name='python.exe'" | Where-Object { $_.CommandLine -match 'herramientas' } | Select-Object ProcessId, CommandLine

# Parar uno por PID (TerminateProcess: cierre en frio, no dispara el "except KeyboardInterrupt"
# del script como si fuera Ctrl+C; los locks huerfanos se limpian solos al reiniciar la moneda)
Stop-Process -Id <PID>
